cmake_minimum_required(VERSION 3.16)
# The app version comes only from src/win/version.h.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/win/version.h" _cr_ver_h)
foreach(_p MAJOR MINOR PATCH)
  string(REGEX MATCH "CR_VERSION_${_p}[ \t]+([0-9]+)" _m "${_cr_ver_h}")
  set(CR_VERSION_${_p} ${CMAKE_MATCH_1})
endforeach()
project(CastRecorder VERSION ${CR_VERSION_MAJOR}.${CR_VERSION_MINOR}.${CR_VERSION_PATCH} LANGUAGES CXX)
if(WIN32)
  enable_language(RC)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_CXX_STANDARD 20)  # C++/WinRT for Windows Graphics Capture
endif()

add_library(cr_core STATIC src/core/hotkey.cpp src/core/settings.cpp src/core/micfx.cpp)
target_include_directories(cr_core PUBLIC src/core)

enable_testing()
add_executable(cr_tests tests/test_main.cpp)
target_link_libraries(cr_tests PRIVATE cr_core)
add_test(NAME unit COMMAND cr_tests WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
# QA edge cases (Nbc): malformed hotkeys/JSON, surrogates, invalid UTF-8, huge inputs.
add_executable(cr_qa_edge tests/qa_edge.cpp)
target_link_libraries(cr_qa_edge PRIVATE cr_core)
add_test(NAME qa_edge COMMAND cr_qa_edge WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
add_executable(cr_fsm tests/test_fsm.cpp)
add_test(NAME fsm COMMAND cr_fsm WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
# Stuck-Finalize hand-off (single publisher, abandon poisons new sessions).
find_package(Threads REQUIRED)
add_executable(cr_handoff tests/test_handoff.cpp)
target_link_libraries(cr_handoff PRIVATE Threads::Threads)
add_test(NAME handoff COMMAND cr_handoff WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})

if(WIN32)
  add_executable(CastRecorder WIN32 src/win/main.cpp src/win/recorder.cpp src/win/hwenc.cpp src/win/preview.cpp src/win/audio.cpp src/win/applog.cpp src/win/windows_enum.cpp src/win/capture_wgc.cpp res/app.rc)
  target_compile_definitions(CastRecorder PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(CastRecorder PRIVATE cr_core d3d11 dxgi mfplat mfreadwrite mfuuid ole32 shell32 shlwapi uuid dwmapi user32 gdi32 winmm ole32 gdiplus)
  if(MSVC)
    target_compile_definitions(CastRecorder PRIVATE CR_HAVE_WGC)
    target_compile_options(CastRecorder PRIVATE /W3 /permissive- /utf-8)
    target_compile_options(cr_core PRIVATE /utf-8)
    target_compile_options(cr_tests PRIVATE /utf-8)
    target_link_libraries(CastRecorder PRIVATE windowsapp)
    target_link_options(CastRecorder PRIVATE /MANIFEST:NO)  # res/app.manifest is embedded via app.rc
    # Hardening: Control Flow Guard, and resolve the exe's static DLL imports from System32 only.
    target_compile_options(CastRecorder PRIVATE /guard:cf)
    target_link_options(CastRecorder PRIVATE /guard:cf /DEPENDENTLOADFLAG:0x800 /DYNAMICBASE /HIGHENTROPYVA /NXCOMPAT)
    # Static CRT: no VC++ redistributable needed on the target PC.
    target_compile_options(cr_qa_edge PRIVATE /utf-8)
    # cr_fsm is header-only logic, no special flags
    set_property(TARGET CastRecorder cr_core cr_tests cr_qa_edge cr_handoff PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_compile_options(CastRecorder PRIVATE -Wall -municode)
    target_link_options(CastRecorder PRIVATE -municode -static -static-libgcc -static-libstdc++)
    target_link_options(cr_tests PRIVATE -static)
    target_link_options(cr_qa_edge PRIVATE -static)
    target_link_options(cr_handoff PRIVATE -static)
    target_link_options(cr_fsm PRIVATE -static)
  endif()
endif()
